# Source: sanshain/templates/backup-cronjob.yaml
apiVersion: batch/v1
kind: CronJob
metadata:
  name: sanshain-backup
  labels:
    helm.sh/chart: sanshain-2.10.0
    app.kubernetes.io/name: sanshain
    app.kubernetes.io/instance: sanshain
    app.kubernetes.io/version: "2.10.0"
    app.kubernetes.io/managed-by: Helm
spec:
  schedule: "0 2 * * *"
  concurrencyPolicy: Forbid
  successfulJobsHistoryLimit: 3
  failedJobsHistoryLimit: 3
  jobTemplate:
    spec:
      template:
        metadata:
          labels:
            app.kubernetes.io/name: sanshain
            app.kubernetes.io/instance: sanshain
            app.kubernetes.io/component: backup
        spec:
          restartPolicy: OnFailure
          securityContext:
            fsGroup: 1000
          containers:
            - name: backup
              securityContext:
                runAsNonRoot: true
                runAsUser: 1000
              env:
                - name: RETAIN
                  value: "7"
                - name: DATABASE_URL
                  valueFrom:
                    secretKeyRef:
                      name: sanshain-secrets
                      key: DATABASE_URL
              image: "postgres:17-alpine"
              command: ["/bin/sh", "-ec"]
              args:
                - |
                  rm -f /backup/*.part
                  out="/backup/sanshain-$(date -u +%Y%m%dT%H%M%SZ).dump"
                  pg_dump --format=custom --file="$out.part" "$DATABASE_URL"
                  mv "$out.part" "$out"
                  ls -1t /backup/*.dump | tail -n +$((RETAIN + 1)) | xargs -r rm -f
                  ls -l /backup
              volumeMounts:
                - name: backup
                  mountPath: /backup
          volumes:
            - name: backup
              persistentVolumeClaim:
                claimName: sanshain-backup
---
